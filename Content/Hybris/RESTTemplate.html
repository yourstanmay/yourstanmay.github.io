------------Integrating REST API using RestTemplate----------

        What is RestTemplate?
            RestTemplate is a core utility class provided by the Spring Framework that 
            allows your Java application to make HTTP requests (like GET, POST, PUT, DELETE) to external web services. 
            It abstracts away low-level HTTP client code, making it easy to send requests and 
            parse JSON responses directly into Java objects (POJOs).

        What is it used for?
            In SAP Commerce Cloud (Hybris), RestTemplate is commonly used to communicate with third-party systems such as 
            payment gateways, wallet engines, loyalty programs, or inventory systems. 
            It handles critical integration tasks like passing authentication headers (Basic Auth or Bearer tokens), 
            handling timeouts, and mapping external API responses into your custom services.

        package com.yourstanmay.controllers;

        import com.yourstanmay.facades.customer.data.WalletDetailsData;
        import com.yourstanmay.facades.customer.wallet.WalletFacade;
        import com.yourstanmay.occ.dto.customer.WalletDetailsWsDTO;
        import de.hybris.platform.servicelayer.user.UserService;
        import de.hybris.platform.webservicescommons.mapping.DataMapper;
        import de.hybris.platform.webservicescommons.swagger.ApiBaseSiteIdAndUserIdParam;

        import de.hybris.platform.webservicescommons.swagger.ApiFieldsParam;
        import org.slf4j.Logger;
        import org.slf4j.LoggerFactory;

        import org.springframework.beans.factory.annotation.Autowired;
        import org.springframework.stereotype.Controller;
        import org.springframework.web.bind.annotation.*;

        @Controller
        @RequestMapping(value = "/{baseSiteId}/users/{userId}/wallet")
        public class YTSWalletController {

            private static final Logger LOG = LoggerFactory.getLogger(YTSWalletController.class);
            private static final String DEFAULT_FIELD_SET = "DEFAULT";

            @Autowired
            private WalletFacade walletFacade;

            @Autowired
            private DataMapper dataMapper;

            @Autowired
            private UserService userService;

            @Secured({ "ROLE_CUSTOMERGROUP", "ROLE_TRUSTED_CLIENT", "ROLE_CUSTOMERMANAGERGROUP" })
            @GetMapping
            @ResponseBody
            @ApiBaseSiteIdAndUserIdParam
            public WalletDetailsWsDTO getWalletDetails(
                    @ApiFieldsParam @RequestParam(defaultValue = DEFAULT_FIELD_SET) final String fields) {

                LOG.info("Received OCC request to fetch wallet details for userId: {}",
                        userService.getCurrentUser().getUid());

                final WalletDetailsData walletData = walletFacade.getWalletDetails(userService.getCurrentUser().getUid());

                if (walletData == null) {
                    return null;
                }

                // Map internal Data Object (WalletDetailsData) to Web Service DTO (WalletDetailsWsDTO)
                return dataMapper.map(walletData, WalletDetailsWsDTO.class, fields);
            }

        }

        package com.yourstanmay.core.customer.wallet.impl;

        import com.yourstanmay.core.customer.wallet.WalletService;
        import com.yourstanmay.core.customer.wallet.dao.BasicCredentialDao;
        import com.yourstanmay.facades.customer.data.WalletDetailsData;
        import de.hybris.platform.apiregistryservices.model.BasicCredentialModel;
        import de.hybris.platform.servicelayer.config.ConfigurationService;
        import org.slf4j.Logger;
        import org.slf4j.LoggerFactory;
        import org.springframework.beans.factory.annotation.Autowired;
        import org.springframework.beans.factory.annotation.Qualifier;
        import org.springframework.http.*;
        import org.springframework.web.client.HttpClientErrorException;
        import org.springframework.web.client.HttpServerErrorException;
        import org.springframework.web.client.RestTemplate;

        import java.nio.charset.StandardCharsets;
        import java.util.Base64;
        import java.util.Collections;

        public class WalletServiceImpl implements WalletService {

            private static final Logger LOG = LoggerFactory.getLogger(WalletServiceImpl.class);

            // Configuration Keys
            private static final String API_URL_KEY = "external.wallet.api.url";
            private static final String CREDENTIAL_ID_KEY = "external.wallet.api.credential.id";

            @Autowired
            @Qualifier("walletRestTemplate")
            private RestTemplate restTemplate;
            @Autowired
            private BasicCredentialDao basicCredentialDao;
            @Autowired
            private ConfigurationService configurationService;

            @Override
            public WalletDetailsData fetchWalletDetails(final String userId) {
                // 1. Fetch values dynamically from ConfigurationService
                final String baseUrl = configurationService.getConfiguration().getString(API_URL_KEY);
                final String credentialId = configurationService.getConfiguration().getString(CREDENTIAL_ID_KEY);

                final String endpoint = String.format("%s/users/%s/wallet", baseUrl, userId);
                // 1. Fetch BasicCredential record from Backoffice using credentialId
                final BasicCredentialModel credential = basicCredentialDao.getBasicCredentialById(credentialId);
                if (credential == null) {
                    LOG.error("BasicCredential model with ID [{}] not found in Backoffice!", credentialId);
                    return null;
                }

                // 2. Build HTTP Headers with dynamic Base64 Basic Auth
                final HttpHeaders headers = new HttpHeaders();
                headers.setAccept(Collections.singletonList(MediaType.APPLICATION_JSON));
                headers.setContentType(MediaType.APPLICATION_JSON);

                final String authHeader = createBasicAuthHeader(credential.getUsername(), credential.getPassword());
                headers.set("Authorization", authHeader);

                // 3. Construct HTTP Request Entity
                final HttpEntity<Void> requestEntity = new HttpEntity<>(headers);
                WalletDetailsData walletDetailsData = null;
                try {
                    LOG.info("Executing GET request to external Wallet API: {}", endpoint);

                    final ResponseEntity<WalletDetailsData> responseEntity = restTemplate.exchange(
                            endpoint,
                            HttpMethod.GET,
                            requestEntity,
                            WalletDetailsData.class
                    );

                    if (responseEntity.getStatusCode().is2xxSuccessful() && responseEntity.hasBody()) {
                        LOG.info("Successfully retrieved wallet details for userId: {}", userId);
                        return responseEntity.getBody();
                    }

                } catch (HttpClientErrorException | HttpServerErrorException e) {
                    LOG.error("HTTP error during Wallet API call. Status: {}, Response: {}",
                            e.getStatusCode(), e.getResponseBodyAsString(), e);
                    walletDetailsData = new WalletDetailsData();
                    walletDetailsData.setError(e.getResponseBodyAsString());
                } catch (Exception e) {
                    LOG.error("Unexpected error fetching wallet details for userId: {}", userId, e);
                }

                return walletDetailsData;
            }

            /**
            * Helper to encode username and password into "Basic <Base64String>"
            */
            private String createBasicAuthHeader(final String username, final String password) {
                final String authString = username + ":" + password;
                final byte[] encodedAuth = Base64.getEncoder().encode(authString.getBytes(StandardCharsets.UTF_8));
                return "Basic " + new String(encodedAuth, StandardCharsets.UTF_8);
            }

        }

        package com.yourstanmay.core.customer.wallet.dao.impl;

        import com.yourstanmay.core.customer.wallet.dao.BasicCredentialDao;
        import de.hybris.platform.apiregistryservices.model.BasicCredentialModel;
        import de.hybris.platform.servicelayer.search.FlexibleSearchQuery;
        import de.hybris.platform.servicelayer.search.FlexibleSearchService;
        import org.apache.commons.collections4.CollectionUtils;
        import org.springframework.beans.factory.annotation.Autowired;

        import java.util.List;

        public class BasicCredentialDaoImpl implements BasicCredentialDao {

            @Autowired
            private FlexibleSearchService flexibleSearchService;

            /**
            * Query Backoffice BasicCredential table by ID using FlexibleSearchService
            */
            @Override
            public BasicCredentialModel getBasicCredentialById(final String id) {
                final String query = "SELECT {" + BasicCredentialModel.PK + "} FROM {" + BasicCredentialModel._TYPECODE
                        + "} WHERE {" + BasicCredentialModel.ID + "} = ?credentialId";

                final FlexibleSearchQuery searchQuery = new FlexibleSearchQuery(query);
                searchQuery.addQueryParameter("credentialId", id);

                final List<BasicCredentialModel> results = flexibleSearchService.<BasicCredentialModel>search(searchQuery).getResult();
                return CollectionUtils.isNotEmpty(results) ? results.get(0) : null;
            }
        }
