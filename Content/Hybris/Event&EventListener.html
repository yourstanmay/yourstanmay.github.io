------------Event & EventListener IN SAP HYBRIS----------

        Event
            A signal in SAP Commerce that announces something just happened (e.g., an order was placed).
            Use: Carries contextual data and notifies the rest of the system about the change.

        Event Listener
            A background component that listens for specific events and executes code when they occur.
            Use: Performs automated follow-up tasks—like sending emails or syncing inventory—without delaying the user.

        Why Use Them?
            Keeps code clean: Decouples core business logic from side tasks.
            Improves performance: Runs heavy tasks asynchronously in the background.

        package com.yourstanmay.core.event.custom.eventlistener;

        import com.yourstanmay.core.event.custom.LoyaltyPointsEvent;
        import de.hybris.platform.core.model.order.OrderModel;
        import de.hybris.platform.core.model.user.CustomerModel;
        import de.hybris.platform.servicelayer.event.impl.AbstractEventListener;
        import de.hybris.platform.servicelayer.model.ModelService;
        import org.slf4j.Logger;
        import org.slf4j.LoggerFactory;
        import org.springframework.beans.factory.annotation.Autowired;

        public class LoyaltyPointsEventListener extends AbstractEventListener<LoyaltyPointsEvent> {

            private static final Logger LOG = LoggerFactory.getLogger(LoyaltyPointsEventListener.class);

            @Autowired
            ModelService modelService;

            @Override
            protected void onEvent(final LoyaltyPointsEvent event) {
                final OrderModel order = event.getOrder();
                final String orderCode = order.getCode();
                final String threadName = Thread.currentThread().getName();

                LOG.info("Started Loyalty Points calculation for Order [{}] on Background Thread [{}]", orderCode, threadName);

                try {
                    // Simulate heavy business logic / External API delay (e.g., 3 seconds)
                    Thread.sleep(3000);

                    // 1. Calculate Points based on Order Total
                    double pointsEarned = calculatePoints(order);

                    // 2. Update Customer's Loyalty Balance in DB or External Loyalty Engine
                    updateCustomerWallet(order.getUser().getUid(), pointsEarned);

                    CustomerModel customerModel = (CustomerModel) order.getUser();
                    customerModel.setLoyaltyPoint(customerModel.getLoyaltyPoint() + pointsEarned);
                    modelService.save(order.getUser());

                    LOG.info("Successfully awarded {}.0f loyalty points for Order [{}] on Thread [{}]",
                            pointsEarned, orderCode, threadName);

                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    LOG.error("Loyalty calculation thread interrupted for order: {}", orderCode, e);
                } catch (Exception e) {
                    // Fault Isolation: Main order placement remains unaffected!
                    LOG.error("Failed to calculate loyalty points for order: {}", orderCode, e);
                }
            }

            private double calculatePoints(OrderModel order) {
                // Business Logic: 10% of total price
                return order.getTotalPrice() / 100 * 10;
