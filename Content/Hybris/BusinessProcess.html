------------ BusinessProcess IN SAP HYBRIS ----------

        A Business Process is an automated workflow engine that executes a sequence of background tasks in SAP Commerce.
        It automates core business operations like order fulfillment, payment processing, external integration, 
        and email delivery asynchronously.

        package com.yourstanmay.fulfilmentprocess.actions.order.custom;

        import com.yourstanmay.core.enums.ErpOrderStatus;
        import com.yourstanmay.core.model.ErpOrderProcessModel;
        import de.hybris.platform.orderprocessing.model.OrderProcessModel;
        import de.hybris.platform.processengine.BusinessProcessService;
        import de.hybris.platform.processengine.action.AbstractProceduralAction;
        import org.slf4j.Logger;
        import org.slf4j.LoggerFactory;
        import org.springframework.beans.factory.annotation.Autowired;

        public class StartErpOrderAction extends AbstractProceduralAction<OrderProcessModel> {

            private static final Logger LOG = LoggerFactory.getLogger(StartErpOrderAction.class);

            @Autowired
            private BusinessProcessService businessProcessService;

            @Override
            public void executeAction(final OrderProcessModel process) {
                final ErpOrderProcessModel erpOrderProcessModel = businessProcessService
                        .createProcess("erpOrderProcess" + process.getOrder().getCode() + "-" + System.currentTimeMillis(), "erpOrderProcess");
                if (LOG.isInfoEnabled())
                {
                    LOG.info("Erp Business Process Starting: {} in step {}", erpOrderProcessModel.getCode(), getClass());
                }
                erpOrderProcessModel.setOrder(process.getOrder());
                erpOrderProcessModel.setOrderID(process.getOrder().getCode());
                erpOrderProcessModel.setErpOrderStatus(ErpOrderStatus.ERP_ORDER_PROCESSING);

                modelService.save(erpOrderProcessModel);
                businessProcessService.startProcess(erpOrderProcessModel);
            }

        }

            <end id="success" state="SUCCEEDED">ERP order integration completed successfully.</end>

        </process>

        public class SendOrderToErpAction extends AbstractSimpleDecisionAction<ErpOrderProcessModel> {

            private static final Logger LOG = LoggerFactory.getLogger(SendOrderToErpAction.class);

            @Override
            public Transition executeAction(final ErpOrderProcessModel process) {
                final OrderModel order = process.getOrder();
                LOG.info("Sending Order {} to ERP system...", order.getCode());

                try {
                    boolean isSent = callErpRestApi(order);
                    if (isSent) {
                        LOG.info("Order payload sent successfully. Transitioning to WAIT state.");
                        return Transition.OK;
                    }
                } catch (Exception e) {
                    LOG.error("Failed to send order to ERP", e);
                }

                return Transition.NOK;
            }

            }
        }

        package com.yourstanmay.fulfilmentprocess.actions.order.custom;

        import com.yourstanmay.core.enums.ErpOrderStatus;
        import com.yourstanmay.core.model.ErpOrderProcessModel;
        import de.hybris.platform.processengine.action.AbstractSimpleDecisionAction;
        import org.apache.log4j.Logger;

        public class ProcessErpAckAction extends AbstractSimpleDecisionAction<ErpOrderProcessModel> {

            private static final Logger LOG = Logger.getLogger(ProcessErpAckAction.class);

            @Override
            public Transition executeAction(final ErpOrderProcessModel process) {
                LOG.info("Processing ERP Ack for process: " + process.getCode());

                String responseCode = process.getErpResponseCode();

                if ("200".equals(responseCode) || "SUCCESS".equalsIgnoreCase(responseCode)) {
                    process.getOrder().setErpOrderStatus(ErpOrderStatus.ERP_CONFIRMED);
                    process.setErpOrderStatus(ErpOrderStatus.ERP_CONFIRMED);
                    modelService.saveAll(process, process.getOrder());
                    LOG.info("Order " + process.getOrder().getCode() + " set to ERP_CONFIRMED");
                    return Transition.OK;
                }

                process.getOrder().setErpOrderStatus(ErpOrderStatus.ERP_REJECTED);
                process.setErpOrderStatus(ErpOrderStatus.ERP_REJECTED);
                modelService.saveAll(process, process.getOrder());
                return Transition.NOK;
            }

        }

        import org.springframework.http.ResponseEntity;
        import org.springframework.stereotype.Controller;
        import org.springframework.web.bind.annotation.PostMapping;
        import org.springframework.web.bind.annotation.RequestMapping;
        import org.springframework.web.bind.annotation.RequestParam;

        @Controller
        @RequestMapping(value = "/{baseSiteId}/erp/webhook")
        @CacheControl(directive = CacheControlDirective.NO_CACHE)
        @Tag(name = "YoursTanmayShopping ERP Webhook")
        public class YTSErpWebhookController {
            private static final Logger LOG = LoggerFactory.getLogger(YTSErpWebhookController.class);
            @Autowired
            private ErpOrderProcessFacade erpOrderProcessFacade;

            @PostMapping("/callback")
            public ResponseEntity<String> receiveErpAck(@RequestParam String processCode,
                                                        @RequestParam String statusCode) {
                LOG.info("Webhook -> Receive ERP Order Ack for process code {}", processCode);

                if (erpOrderProcessFacade.processErpOrder(processCode, statusCode)) {
                    return ResponseEntity.ok("Process resumed successfully.");
                }
                return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).build();
            }


        }

            }

        package com.yourstanmay.core.erp.impl;

        import com.yourstanmay.core.erp.ErpOrderProcessService;
        import com.yourstanmay.core.model.ErpOrderProcessModel;
        import de.hybris.platform.processengine.BusinessProcessService;
        import de.hybris.platform.servicelayer.model.ModelService;
        import org.slf4j.Logger;
        import org.slf4j.LoggerFactory;
        import org.springframework.beans.factory.annotation.Autowired;

        public class ErpOrderProcessServiceImpl implements ErpOrderProcessService {

            private static final Logger LOG = LoggerFactory.getLogger(ErpOrderProcessServiceImpl.class);

            @Autowired
            private BusinessProcessService businessProcessService;
            @Autowired
            private ModelService modelService;

            @Override
            public void startErpBusinessProcess(final String processCode, final String statusCode) {
                // 1. Fetch paused process instance
                ErpOrderProcessModel process = businessProcessService.getProcess(processCode);
                process.setErpResponseCode(statusCode);
                process.getOrder().setErpResponseCode(statusCode);
                modelService.saveAll(process, process.getOrder());

                // 2. Trigger event matching the <event> tag in erpOrderProcess.xml
                businessProcessService.triggerEvent(processCode + "_ERP_ACK");
                LOG.info("Process {} has been started via event", processCode);
            }

        }

        ---------- Difference between Cronjob & Business Process ----------

        CronJob schedules and runs background tasks at specific times, 
        whereas Business Process manages the end-to-end workflow of a business operation such as order fulfillment or email sending.

        CronJob = “When to run”, 
        Business Process = “What business steps to execute.”

        :.............................................:...................................................................:
        :                    CronJob                  :                     Business Process                              :
        :.............................................:...................................................................:
        : Runs a task automatically at a scheduled    : Executes a sequence of business steps                             :
        : time                                        :                                                                   :
        :.............................................:...................................................................:
        : Like a scheduler/alarm                      : Like a workflow/process                                           :
        :.............................................:...................................................................:
        : Usually independent                         : Triggered by a business event (order placed, payment              :
        :                                             : success, etc.)                                                    :
        :.............................................:...................................................................:
        : Can run every hour/day/week                 : Runs until all defined steps are completed                        :
        :.............................................:...................................................................:
        : Example: product indexing, stock sync,      : Example: order confirmation, payment, consignment, shipping,      :
        : cleanup                                     : email                                                             :
        :.............................................:...................................................................:
