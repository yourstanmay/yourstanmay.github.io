------------🕒CronJob IN SAP HYBRIS----------

        CronJob ->  A CronJob is a scheduled background task in SAP Hybris that runs automatically at a specified time or interval, without manual intervention.
	        Use-> To automate repetitive tasks like generating reports, syncing data, sending emails, or cleaning up old records.

        Trigger->  A Trigger defines when a CronJob should execute - it holds the schedule (cron expression) for the job.
	        Use-> To control the execution timing of a CronJob (e.g., daily, hourly, or at a specific time). Without a Trigger, a CronJob will not run on its own.

        Composite CronJob-> A Composite CronJob is a CronJob that groups multiple individual CronJobs together and runs them in a defined sequence, one after another.
	        Use-> To chain dependent jobs together (e.g., Catalog Sync → Solr Indexing → Report Generation) so they execute in the correct order automatically, 
                     instead of managing separate triggers for each.

        package com.yourstanmay.core.job;

        import de.hybris.platform.basecommerce.enums.InStockStatus;
        import de.hybris.platform.catalog.CatalogVersionService;
        import de.hybris.platform.catalog.model.CatalogVersionModel;
        import de.hybris.platform.cronjob.enums.CronJobResult;
        import de.hybris.platform.cronjob.enums.CronJobStatus;
        import de.hybris.platform.cronjob.model.CronJobModel;
        import de.hybris.platform.servicelayer.cronjob.AbstractJobPerformable;
        import de.hybris.platform.servicelayer.cronjob.PerformResult;
        import de.hybris.platform.servicelayer.search.FlexibleSearchQuery;
        import de.hybris.platform.servicelayer.search.SearchResult;
        import org.slf4j.Logger;
        import org.slf4j.LoggerFactory;
        import org.springframework.beans.factory.annotation.Autowired;

        import java.util.Collections;
        import java.util.HashMap;
        import java.util.List;
        import java.util.Map;

        public class ProductReportJobPerformable extends AbstractJobPerformable<CronJobModel> {

            private static final Logger LOG = LoggerFactory.getLogger(ProductReportJobPerformable.class);

            private static final String CATALOG_ID = "ytsProductCatalog";
            private static final String STAGED_VERSION = "Staged";
            private static final String ONLINE_VERSION = "Online";

            @Autowired
            private CatalogVersionService catalogVersionService;

            @Override
            public PerformResult perform(final CronJobModel cronJob) {

                //====================== CronJob Business Logic - Start ==========================
                LOG.info("Starting Product Summary Report CronJob...");

                // 1. Fetch Staged & Online Catalog Versions
                CatalogVersionModel stagedVersion = catalogVersionService.getCatalogVersion(CATALOG_ID, STAGED_VERSION);
                CatalogVersionModel onlineVersion = catalogVersionService.getCatalogVersion(CATALOG_ID, ONLINE_VERSION);

                // 2. Query Total Products in Staged & Online
                int stagedCount = getProductCountByCatalogVersion(stagedVersion);
                int onlineCount = getProductCountByCatalogVersion(onlineVersion);

                // 3. Fetch Product Codes (IDs) that are Out of Stock
                List<String> outOfStockProductCodes = getOutOfStockProductCodes();

                // 4. Output Summary Log Report
                LOG.info("=================================================");
                LOG.info("             PRODUCT SUMMARY REPORT              ");
                LOG.info("=================================================");
                LOG.info(" Total Products in STAGED Catalog : {}", stagedCount);
                LOG.info(" Total Products in ONLINE Catalog : {}", onlineCount);
                LOG.info(" Total Products OUT OF STOCK      : {}", outOfStockProductCodes.size());

                if (!outOfStockProductCodes.isEmpty()) {
                    LOG.info(" Out of Stock Product IDs         : {}", String.join(", ", outOfStockProductCodes));
                } else {
                    LOG.info(" Out of Stock Product IDs         : None");
                }

                LOG.info("=================================================");

                //====================== CronJob Business Logic - End ==========================

                return new PerformResult(CronJobResult.SUCCESS, CronJobStatus.FINISHED);
            }

            private int getProductCountByCatalogVersion(CatalogVersionModel catalogVersion) {
                String query = "SELECT COUNT({pk}) FROM {Product} WHERE {catalogVersion} = ?cv";
                Map<String, Object> params = Collections.singletonMap("cv", catalogVersion);
                FlexibleSearchQuery searchQuery = new FlexibleSearchQuery(query, params);
                // Explicitly set the result class to Integer so Hybris does NOT treat the count as a Model PK
                searchQuery.setResultClassList(Collections.singletonList(Integer.class));
                SearchResult<Integer> result = flexibleSearchService.search(searchQuery);
                if (result.getResult() != null && !result.getResult().isEmpty()) {
                    return result.getResult().get(0);
                }
                return 0;
            }

            private List<String> getOutOfStockProductCodes() {
                String query = "SELECT DISTINCT {s.productCode} FROM {StockLevel AS s} "
                        + "WHERE {s.available} <= 0 OR {s.inStockStatus} = ?forceStatus";
                Map<String, Object> params = new HashMap<>();
                params.put("forceStatus", InStockStatus.FORCEOUTOFSTOCK);
                FlexibleSearchQuery searchQuery = new FlexibleSearchQuery(query, params);
                searchQuery.setResultClassList(Collections.singletonList(String.class));
                SearchResult<String> result = flexibleSearchService.search(searchQuery);
                return result.getResult();
            }
        }

        package com.yourstanmay.core.job;

        import com.yourstanmay.core.model.OrderReportCronJobModel;
        import de.hybris.platform.core.model.order.OrderModel;
        import de.hybris.platform.cronjob.enums.CronJobResult;
        import de.hybris.platform.cronjob.enums.CronJobStatus;
        import de.hybris.platform.servicelayer.cronjob.AbstractJobPerformable;
        import de.hybris.platform.servicelayer.cronjob.PerformResult;
        import de.hybris.platform.servicelayer.search.FlexibleSearchQuery;
        import de.hybris.platform.servicelayer.search.SearchResult;
        import org.slf4j.Logger;
        import org.slf4j.LoggerFactory;

        import java.util.*;

        public class OrderReportJobPerformable extends AbstractJobPerformable<OrderReportCronJobModel> {

            private static final Logger LOG = LoggerFactory.getLogger(OrderReportJobPerformable.class);

            @Override
            public PerformResult perform(final OrderReportCronJobModel cronJob) {

                //====================== CronJob Business Logic - Start ==========================

                // Get configured report duration (default to 1 if not specified or invalid)
                int reportDays = (cronJob.getReportDays() != null && cronJob.getReportDays() > 0)
                        ? cronJob.getReportDays()
                        : 1;

                LOG.info("Starting Order & Profit Summary CronJob for the past {} day(s)...", reportDays);

                // 1. Calculate End of Day (Today 23:59:59)
                Calendar cal = Calendar.getInstance();
                cal.set(Calendar.HOUR_OF_DAY, 23);
                cal.set(Calendar.MINUTE, 59);
                cal.set(Calendar.SECOND, 59);
                cal.set(Calendar.MILLISECOND, 999);
                Date endOfDay = cal.getTime();

                // 2. Calculate Start Date depending on reportDays (Subtracting N - 1 days)
                // e.g., if reportDays = 1, start date is today at 00:00:00
                // e.g., if reportDays = 3, start date is 2 days ago at 00:00:00
                cal.add(Calendar.DAY_OF_MONTH, -(reportDays - 1));
                cal.set(Calendar.HOUR_OF_DAY, 0);
                cal.set(Calendar.MINUTE, 0);
                cal.set(Calendar.SECOND, 0);
                cal.set(Calendar.MILLISECOND, 0);
                Date startOfDay = cal.getTime();

                // 3. Fetch orders within the date range via FlexibleSearch
                String query = "SELECT {pk} FROM {Order} WHERE {creationtime} >= ?startOfDay AND {creationtime} <= ?endOfDay";
                Map<String, Object> params = new HashMap<>();
                params.put("startOfDay", startOfDay);
                params.put("endOfDay", endOfDay);

                SearchResult<OrderModel> searchResult = flexibleSearchService.search(new FlexibleSearchQuery(query, params));
                List<OrderModel> orders = searchResult.getResult();

                int totalOrdersCount = orders.size();
                double totalRevenue = 0.0;

                // 4. Aggregate totals
                for (OrderModel order : orders) {
                    if (order.getTotalPrice() != null) {
                        totalRevenue += order.getTotalPrice();
                    }
                }
                // 5. Output Report Summary
                LOG.info("=================================================");
                LOG.info(" ORDER REPORT [{} DAY(S): {} to {}]", reportDays, startOfDay, endOfDay);
                LOG.info("=================================================");
                LOG.info(" Total Orders Placed : {}", totalOrdersCount);
                LOG.info(" Total Revenue Amount: ${}", String.format("%.2f", totalRevenue));
                LOG.info("=================================================");

                //====================== CronJob Business Logic - End ==========================

                return new PerformResult(CronJobResult.SUCCESS, CronJobStatus.FINISHED);
            }


        }

        ────────────────────────────────────────────────
        CRON EXPRESSION FORMAT FOR TRIGGER (SAP Commerce)
        ────────────────────────────────────────────────
        seconds  minutes  hours  day-of-month  month  day-of-week  [year]

        FIELD           ALLOWED VALUES        SPECIAL CHARACTERS
        ────────────────────────────────────────────────────────
        Seconds         0-59                  , - * /
        Minutes         0-59                  , - * /
        Hours           0-23                  , - * /
        Day-of-month    1-31                  , - * / ? L W
        Month           1-12 or JAN-DEC       , - * /
        Day-of-week     1-7 or SUN-SAT        , - * / ? L #
        Year (optional) empty, 1970-2099      , - * /

        SPECIAL CHARACTERS
        ────────────────────────────────────────────────
        *   every value
        ?   no specific value (day-of-month OR day-of-week,
            when the other is specified — do not specify both)
        -   range                  e.g. MON-FRI
        ,   list                   e.g. MON,WED,FRI
        /   increments             e.g. 0/15 = every 15 min from 0
        L   last day of month / last weekday
        W   nearest weekday to a given day
        #   nth weekday of month   e.g. 6#3 = third Friday

        EXAMPLES
        ────────────────────────────────────────────────
        0 0 2 * * ?           → Every day at 2:00:00 AM
        0 15 10 ? * *         → Every day at 10:15 AM
        0 0 12 * * ?           → Every day at noon
        0 0/30 8-17 * * ?      → Every 30 min, 8 AM-5 PM, daily
        0 0 0 1 * ?            → Midnight on the 1st of every month
        0 0 6 ? * MON-FRI      → 6:00 AM every weekday
        0 0 0 ? * 1            → Midnight every Sunday
        0 30 9 15 * ?          → 9:30 AM on the 15th of every month
        0 0 0 L * ?            → Midnight on the last day of every month
        0 0 2 ? * SAT,SUN      → 2:00 AM every Saturday and Sunday
