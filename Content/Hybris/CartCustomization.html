------------🛒 CART CUSTOMIZATION IN SAP HYBRIS----------

    ###################################################################
    #### Prevent Cart Entry Merging on Add-to-Cart in SAP Commerce ####
    ###################################################################

    By default, SAP Commerce merges duplicate products into a single line item by incrementing its quantity. 
    To create a separate cart entry for every add-to-cart action, 
    you can use 3 different approaches:
        Approach 1: Add-to-Cart Hook (CommerceAddToCartMethodHook) 
                    - Sets createNewEntry = true right before the add-to-cart execution begins.

        Approach 2: Entry Merge Filter (EntryMergeFilter) 
                    - Evaluates candidate cart entries and returns Boolean.FALSE to prevent merging.

        Approach 3: Cart Parameter Populator (Populator) 
                    - Sets createNewEntry = true at the facade level during parameter conversion.

        *********************************************************
                        yourstanmaycore-spring.xml
        *********************************************************

        <!-- ================================================================= -->
        <!-- APPROACH 1: Cart Parameter Populator                                     -->
        <!-- ================================================================= -->
            <bean id="ytsCartParameterPopulator" class="com.yourstanmay.core.cart.populators.YTSCartParameterPopulator"/>

            <bean id="ytsCartParameterConverterMerge"
                depends-on="commerceCartParameterConverter"
                parent="modifyPopulatorList">
                <property name="list" ref="commerceCartParameterConverter"/>
                <property name="add" ref="ytsCartParameterPopulator"/>
            </bean>

        <!-- ================================================================= -->
        <!-- APPROACH 2: Add-To-Cart Hook                                     -->
        <!-- ================================================================= -->
            <!-- Define the Custom Hook Bean -->
            <bean id="addNewCartEntryHook" class="com.yourstanmay.core.cart.hooks.AddNewCartEntryHook" />

            <!-- Register it into the OOTB commerceAddToCartMethodHooks list -->
            <bean id="addNewCartEntryHookMergeDirective"
                depends-on="commerceAddToCartMethodHooks"
                parent="listMergeDirective">
                <property name="add" ref="addNewCartEntryHook" />
            </bean>

        <!-- ================================================================= -->
        <!-- APPROACH 3: Cart Entry Merge Filter                                     -->
        <!-- ================================================================= -->
            <!-- 1. Define the Custom Filter Bean -->
            <bean id="addNewEntryMergeFilter" class="com.yourstanmay.core.cart.filters.AddNewEntryMergeFilter" />

            <!-- 2. Append to OOTB commerceCartEntryMergeFilters list -->
            <bean id="addNewEntryMergeFilterMergeDirective"
                depends-on="commerceCartEntryMergeFilters"
                parent="listMergeDirective">
                <property name="add" ref="addNewEntryMergeFilter" />
            </bean>

        
        ************************ APPROACH 1: Cart Parameter Populator ***********************************
            yourstanmaycore/src/com/yourstanmay/core/cart/populators/MyntraCartParameterPopulator.java
        *************************************************************************************************

        package com.yourstanmay.core.cart.populators;

        import com.yourstanmay.core.constants.YourstanmayCoreConstants;
        import de.hybris.platform.converters.Populator;
        import de.hybris.platform.commercefacades.order.data.AddToCartParams;
        import de.hybris.platform.commerceservices.service.data.CommerceCartParameter;
        import de.hybris.platform.servicelayer.config.ConfigurationService;
        import org.springframework.beans.factory.annotation.Autowired;

        public class YTSCartParameterPopulator implements Populator<AddToCartParams, CommerceCartParameter> {

            @Autowired
            private ConfigurationService configurationService;

            @Override
            public void populate(final AddToCartParams source, final CommerceCartParameter target) {
                if (target != null && configurationService.getConfiguration()
                        .getBoolean(YourstanmayCoreConstants.ADD_NEW_CART_ENTRY_BY_POPULATOR_ENABLED, false)) {
                    // Force new entry creation when converting parameters
                    target.setCreateNewEntry(true);
                }
            }
        }

        ***************************APPROACH 2: Add-To-Cart Hook******************************
            yourstanmaycore/src/com/yourstanmay/core/cart/hooks/AddNewCartEntryHook.java
        *************************************************************************************

        package com.yourstanmay.core.cart.hooks;

        import com.yourstanmay.core.constants.YourstanmayCoreConstants;
        import de.hybris.platform.commerceservices.order.CommerceCartModification;
        import de.hybris.platform.commerceservices.order.CommerceCartModificationException;
        import de.hybris.platform.commerceservices.order.hook.CommerceAddToCartMethodHook;
        import de.hybris.platform.commerceservices.service.data.CommerceCartParameter;
        import de.hybris.platform.servicelayer.config.ConfigurationService;
        import org.springframework.beans.factory.annotation.Autowired;

        public class AddNewCartEntryHook implements CommerceAddToCartMethodHook {

            @Autowired
            private ConfigurationService configurationService;

            @Override
            public void beforeAddToCart(final CommerceCartParameter parameters) throws CommerceCartModificationException
            {
                if (parameters != null && configurationService.getConfiguration()
                        .getBoolean(YourstanmayCoreConstants.ADD_NEW_CART_ENTRY_HOOK_ENABLED, false)) {
                    parameters.setCreateNewEntry(true);
                }
            }

            @Override
            public void afterAddToCart(final CommerceCartParameter parameters, final CommerceCartModification result) throws
                    CommerceCartModificationException
            {
                // Not needed for forcing new entry
            }

        }

        ****************************APPROACH 3: Cart Entry Merge Filter**************************
            yourstanmaycore/src/com/yourstanmay/core/cart/filters/AddNewEntryMergeFilter.java
        *****************************************************************************************
        
            package com.yourstanmay.core.cart.filters;

            import com.yourstanmay.core.constants.YourstanmayCoreConstants;
            import de.hybris.platform.commerceservices.order.EntryMergeFilter;
            import de.hybris.platform.core.model.order.AbstractOrderEntryModel;
            import de.hybris.platform.servicelayer.config.ConfigurationService;
            import jakarta.annotation.Nonnull;
            import org.springframework.beans.factory.annotation.Autowired;

            public class AddNewEntryMergeFilter implements EntryMergeFilter {

                @Autowired
                private ConfigurationService configurationService;

                @Override
                public Boolean apply(@Nonnull AbstractOrderEntryModel candidate, @Nonnull AbstractOrderEntryModel target) {
                    if (configurationService.getConfiguration()
                            .getBoolean(YourstanmayCoreConstants.ADD_NEW_ENTRY_MERGE_FILTER_ENABLED, false)) {
                        // Returning Boolean.FALSE prevents SAP Commerce from merging candidate entries,
                        // forcing a new entry creation on every add-to-cart call.
                        return Boolean.FALSE;
                    }
                    return Boolean.TRUE;
                }
            }

        ******************************************************************************************
            yourstanmaycore/src/com/yourstanmay/core/constants/YourstanmayCoreConstants.java
        ******************************************************************************************
        
        public static final String ADD_NEW_CART_ENTRY_HOOK_ENABLED = "myntra.addnewcartentryhook.enabled";
	    public static final String ADD_NEW_ENTRY_MERGE_FILTER_ENABLED = "myntra.addnewentryfilter.enabled";
	    public static final String ADD_NEW_CART_ENTRY_BY_POPULATOR_ENABLED = "myntra.addnewentrybypopulator.enabled";

            Properties
